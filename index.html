function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "ONLINE", app: "PT Ruang Kreasi Utama POS API" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const requestData = JSON.parse(e.postData.contents);
    const action = requestData.action;
    const payload = requestData.payload;
    
    let result = {};
    
    switch (action) {
      case "getDashboardData":
        result = getDashboardData();
        break;
      case "createTransaction":
        result = createTransactionServer(payload);
        break;
      case "recordPayment":
        result = recordPaymentServer(payload);
        break;
      case "cancelInvoice":
        result = cancelInvoiceServer(payload);
        break;
      default:
        result = { success: false, error: "Action tidak dikenal" };
    }
    
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// SECURE SERVER-SIDE SEQUENTIAL INVOICE GENERATOR
function getNextInvoiceSequence() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("INVOICE_SEQUENCE");
  if (!sheet) {
    sheet = ss.insertSheet("INVOICE_SEQUENCE");
    sheet.appendRow(["year", "current_sequence"]);
  }
  
  const currentYear = new Date().getFullYear();
  const data = sheet.getDataRange().getValues();
  let foundRow = -1;
  let nextSeq = 1;
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == currentYear) {
      foundRow = i + 1;
      nextSeq = data[i][1] + 1;
      break;
    }
  }
  
  if (foundRow > 0) {
    sheet.getRange(foundRow, 2).setValue(nextSeq);
  } else {
    sheet.appendRow([currentYear, 1]);
    nextSeq = 1;
  }
  
  const seqPadded = String(nextSeq).padStart(6, '0');
  const invoiceNum = "RKU-INV-" + currentYear + "-" + seqPadded;
  return invoiceNum;
}
